function startFirebaseListener() { if (typeof firebase === 'undefined') return; const db = firebase.database(); const todayStr = new Date().toISOString().split('T')[0]; db.ref(`contacted_buyers/${todayStr}`).on('value', (snapshot) => { window.contactedBuyers = snapshot.val() || {}; if (document.getElementById('page-pipeline').classList.contains('active')) { renderPipeline(); } }); db.ref('/').on('value', (snapshot) => { const raw = snapshot.val() || {}; store.floor = extractRecords(raw.floor); store.stock = extractRecords(raw.stock); store.coldstore = extractRecords(raw.coldstore || raw.coldStore); store.orders = extractRecords(raw.orders || raw.cart || raw.slips || raw.salesHistory); const rawPrefs = raw.preferences || raw.buyerPreferences || {}; const rawPhones = raw.buyerPhones || {}; const buyersNode = raw.buyers || {}; // 1. Build Phone Map with cleaned keys const phoneMap = {}; const phoneKeys = Object.keys(rawPhones); phoneKeys.forEach(key => { const cleanKey = String(key).trim().toUpperCase(); phoneMap[cleanKey] = rawPhones[key]; }); // 2. Smart Phone Lookup Function (Exact + Partial Match) const getPhoneForBuyer = (name, fallback = '') => { if (!name) return fallback || 'No Contact'; const cleanName = String(name).trim().toUpperCase(); // Check 1: Exact match if (phoneMap[cleanName]) return phoneMap[cleanName]; // Check 2: Partial match (e.g., "FERREIRA FRESH" in "FERREIRA FRESH (PTY)") const matchedKey = phoneKeys.find(pk => { const cleanPK = pk.trim().toUpperCase(); return cleanPK.includes(cleanName) || cleanName.includes(cleanPK); }); if (matchedKey && phoneMap[matchedKey.trim().toUpperCase()]) { return phoneMap[matchedKey.trim().toUpperCase()]; } return fallback || 'No Contact'; }; const buyersMap = {}; // Process Preferences Object.keys(rawPrefs).forEach(k => { const cleanKey = String(k).trim().toUpperCase(); buyersMap[cleanKey] = { name: k, phone: getPhoneForBuyer(k), prefs: rawPrefs[k], turnover: 0 }; }); // Process Buyers Node Object.keys(buyersNode).forEach(k => { const b = buyersNode[k] || {}; const bName = b.name || k; const cleanKey = String(bName).trim().toUpperCase(); const existingPhone = buyersMap[cleanKey]?.phone !== 'No Contact' ? buyersMap[cleanKey]?.phone : ''; const phone = getPhoneForBuyer(bName, b.phone || existingPhone); if (buyersMap[cleanKey]) { buyersMap[cleanKey].turnover = b.turnover || b.revenue || buyersMap[cleanKey].turnover; buyersMap[cleanKey].phone = phone; } else { buyersMap[cleanKey] = { name: bName, phone: phone, prefs: [], turnover: b.turnover || b.revenue || 0 }; } }); store.buyers = Object.values(buyersMap); // Refresh ALL Views updateDashboardKPIs(); renderBuyers(); renderStock(); renderFloor(); renderColdstore(); renderOrders(); renderPipeline(); renderCommodities(); }); }
